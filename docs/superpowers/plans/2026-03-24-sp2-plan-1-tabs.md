# SP2 Plan 1: Tab System + Split Panes

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement VS Code-style tabs with full bidirectional split pane support. Users can open multiple requests in tabs, split the editor area horizontally or vertically, drag tabs between groups, and resize panes.

**Architecture:** Recursive `PaneNode` binary tree in Zustand. React component tree mirrors data tree. No Rust backend changes — tabs are purely frontend state.

**Tech Stack:** React 18, TypeScript, Zustand, shadcn/ui (init with `yarn dlx shadcn@latest init --preset b2CkJ2CsV --template vite --monorepo`)

**Prerequisite:** SP2 Plan 0 (bug fixes) must be complete.

```typescript
// Recursive split tree
export type PaneNode = SplitNode | LeafNode;

export interface SplitNode {
  type: 'split';
  id: string;
  direction: 'horizontal' | 'vertical';
  children: [PaneNode, PaneNode];
  sizes: [number, number]; // percentage, e.g. [50, 50]
}

export interface LeafNode {
  type: 'leaf';
  id: string;
  groupId: string;
  tabs: Tab[];
  activeTabId: string;
}

export interface Tab {
  id: string;
  title: string;
  tabType: 'request' | 'draft' | 'history';
  request: RequestState;
  response: ResponseState | null;
  isDirty: boolean;
  source?: { collection: string; path: string };
}

export interface RequestState {
  method: HttpMethod;
  url: string;
  queryParams: KeyValueEntry[];
  headers: KeyValueEntry[];
  body: BodyState;
  auth: AuthState;
}

export interface KeyValueEntry {
  id: string;
  key: string;
  value: string;
  enabled: boolean;
}

export interface BodyState {
  mode: 'none' | 'json' | 'xml' | 'text' | 'formdata' | 'binary';
  content: string;
  formData: KeyValueEntry[];
}

export interface AuthState {
  authType: 'none' | 'basic' | 'bearer' | 'api-key' | 'oauth2' | 'aws-sig-v4';
  basic?: { username: string; password: string };
  bearer?: { token: string };
  apiKey?: { key: string; value: string; addTo: 'header' | 'query' };
  // oauth2 and aws-sig-v4 added in SP2 Plan 3
}

export interface ResponseState {
  status: number;
  statusText: string;
  headers: KeyValueEntry[];
  body: string;
  durationMs: number;
  sizeBytes: number;
  activeView: 'pretty' | 'raw' | 'preview' | 'headers';
}

export type HttpMethod = 'GET' | 'POST' | 'PUT' | 'PATCH' | 'DELETE' | 'OPTIONS' | 'HEAD';
```

```typescript
import { v4 as uuid } from 'uuid';
import type { PaneNode, LeafNode, SplitNode, Tab, RequestState, BodyState, AuthState } from '@/types/pane-types';

export function createDefaultTab(): Tab {
  return {
    id: uuid(),
    title: 'New request',
    tabType: 'draft',
    request: createDefaultRequest(),
    response: null,
    isDirty: false,
  };
}

export function createDefaultRequest(): RequestState {
  return {
    method: 'GET',
    url: '',
    queryParams: [],
    headers: [],
    body: { mode: 'none', content: '', formData: [] },
    auth: { authType: 'none' },
  };
}

export function createDefaultLeaf(groupId?: string): LeafNode {
  const tab = createDefaultTab();
  return {
    type: 'leaf',
    id: uuid(),
    groupId: groupId ?? uuid(),
    tabs: [tab],
    activeTabId: tab.id,
  };
}

// ---- Tree traversal ----

export function findLeaf(node: PaneNode, groupId: string): LeafNode | null {
  if (node.type === 'leaf') {
    return node.groupId === groupId ? node : null;
  }
  return findLeaf(node.children[0], groupId) ?? findLeaf(node.children[1], groupId);
}

export function findTabInTree(node: PaneNode, tabId: string): { leaf: LeafNode; tab: Tab } | null {
  if (node.type === 'leaf') {
    const tab = node.tabs.find((t) => t.id === tabId);
    return tab ? { leaf: node, tab } : null;
  }
  return findTabInTree(node.children[0], tabId) ?? findTabInTree(node.children[1], tabId);
}

export function findFirstLeaf(node: PaneNode): LeafNode {
  if (node.type === 'leaf') return node;
  return findFirstLeaf(node.children[0]);
}

export function findActiveLeaf(node: PaneNode, activeGroupId: string): LeafNode {
  return findLeaf(node, activeGroupId) ?? findFirstLeaf(node);
}

// ---- Immutable updates ----

export function updateLeaf(
  node: PaneNode,
  groupId: string,
  updater: (leaf: LeafNode) => LeafNode,
): PaneNode {
  if (node.type === 'leaf') {
    return node.groupId === groupId ? updater(node) : node;
  }
  return {
    ...node,
    children: [
      updateLeaf(node.children[0], groupId, updater),
      updateLeaf(node.children[1], groupId, updater),
    ],
  };
}

export function removeLeaf(node: PaneNode, groupId: string): PaneNode {
  if (node.type === 'leaf') return node;
  const [left, right] = node.children;

  if (left.type === 'leaf' && left.groupId === groupId) return right;
  if (right.type === 'leaf' && right.groupId === groupId) return left;

  return {
    ...node,
    children: [
      removeLeaf(left, groupId),
      removeLeaf(right, groupId),
    ],
  };
}

export function splitLeaf(
  node: PaneNode,
  groupId: string,
  direction: 'horizontal' | 'vertical',
): PaneNode {
  if (node.type === 'leaf' && node.groupId === groupId) {
    const newLeaf = createDefaultLeaf();
    return {
      type: 'split',
      id: uuid(),
      direction,
      children: [node, newLeaf],
      sizes: [50, 50],
    };
  }
  if (node.type === 'split') {
    return {
      ...node,
      children: [
        splitLeaf(node.children[0], groupId, direction),
        splitLeaf(node.children[1], groupId, direction),
      ],
    };
  }
  return node;
}
```

`frontend/src/stores/pane-store.ts`:
```typescript
import { create } from 'zustand';
import type { PaneNode, LeafNode, Tab, RequestState } from '@/types/pane-types';
import {
  createDefaultLeaf,
  createDefaultTab,
  findLeaf,
  findFirstLeaf,
  findTabInTree,
  updateLeaf,
  removeLeaf,
  splitLeaf,
} from '@/lib/pane-utils';

interface PaneState {
  root: PaneNode;
  activeGroupId: string;

  // Tab actions
  newDraftTab: (groupId?: string) => void;
  openTab: (tab: Tab, groupId?: string) => void;
  closeTab: (tabId: string, groupId: string) => void;
  setActiveTab: (tabId: string, groupId: string) => void;
  moveTab: (tabId: string, fromGroupId: string, toGroupId: string) => void;

  // Split actions
  splitGroup: (groupId: string, direction: 'horizontal' | 'vertical') => void;
  resizePane: (splitId: string, sizes: [number, number]) => void;

  // Request state actions
  updateRequest: (tabId: string, patch: Partial<RequestState>) => void;
  setResponse: (tabId: string, response: any) => void;
  markDirty: (tabId: string) => void;
  markClean: (tabId: string) => void;

  // Utility
  reset: () => void;
}

function createInitialState() {
  const leaf = createDefaultLeaf();
  return { root: leaf as PaneNode, activeGroupId: leaf.groupId };
}

export const usePaneStore = create<PaneState>((set, get) => ({
  ...createInitialState(),

  newDraftTab: (groupId) => {
    const targetGroup = groupId ?? get().activeGroupId;
    const tab = createDefaultTab();
    set((state) => ({
      root: updateLeaf(state.root, targetGroup, (leaf) => ({
        ...leaf,
        tabs: [...leaf.tabs, tab],
        activeTabId: tab.id,
      })),
    }));
  },

  openTab: (tab, groupId) => {
    const targetGroup = groupId ?? get().activeGroupId;
    // Check if already open anywhere
    const existing = findTabInTree(get().root, tab.id);
    if (existing) {
      set((state) => ({
        root: updateLeaf(state.root, existing.leaf.groupId, (leaf) => ({
          ...leaf,
          activeTabId: tab.id,
        })),
        activeGroupId: existing.leaf.groupId,
      }));
      return;
    }
    set((state) => ({
      root: updateLeaf(state.root, targetGroup, (leaf) => ({
        ...leaf,
        tabs: [...leaf.tabs, tab],
        activeTabId: tab.id,
      })),
    }));
  },

  closeTab: (tabId, groupId) => {
    set((state) => {
      const leaf = findLeaf(state.root, groupId);
      if (!leaf) return state;
      const remaining = leaf.tabs.filter((t) => t.id !== tabId);
      if (remaining.length === 0) {
        // Collapse group
        const newRoot = removeLeaf(state.root, groupId);
        const newActive = findFirstLeaf(newRoot);
        return { root: newRoot, activeGroupId: newActive.groupId };
      }
      const newActiveId = leaf.activeTabId === tabId
        ? remaining[Math.max(0, leaf.tabs.findIndex((t) => t.id === tabId) - 1)].id
        : leaf.activeTabId;
      return {
        root: updateLeaf(state.root, groupId, (l) => ({
          ...l,
          tabs: remaining,
          activeTabId: newActiveId,
        })),
      };
    });
  },

  setActiveTab: (tabId, groupId) => {
    set((state) => ({
      root: updateLeaf(state.root, groupId, (leaf) => ({
        ...leaf,
        activeTabId: tabId,
      })),
      activeGroupId: groupId,
    }));
  },

  moveTab: (tabId, fromGroupId, toGroupId) => {
    set((state) => {
      const fromLeaf = findLeaf(state.root, fromGroupId);
      if (!fromLeaf) return state;
      const tab = fromLeaf.tabs.find((t) => t.id === tabId);
      if (!tab) return state;

      let newRoot = updateLeaf(state.root, fromGroupId, (leaf) => {
        const remaining = leaf.tabs.filter((t) => t.id !== tabId);
        return {
          ...leaf,
          tabs: remaining,
          activeTabId: remaining.length > 0 ? remaining[0].id : leaf.activeTabId,
        };
      });

      // Collapse empty source group
      const updatedFrom = findLeaf(newRoot, fromGroupId);
      if (updatedFrom && updatedFrom.tabs.length === 0) {
        newRoot = removeLeaf(newRoot, fromGroupId);
      }

      newRoot = updateLeaf(newRoot, toGroupId, (leaf) => ({
        ...leaf,
        tabs: [...leaf.tabs, tab],
        activeTabId: tab.id,
      }));

      return { root: newRoot, activeGroupId: toGroupId };
    });
  },

  splitGroup: (groupId, direction) => {
    set((state) => ({
      root: splitLeaf(state.root, groupId, direction),
    }));
  },

  resizePane: (splitId, sizes) => {
    set((state) => ({
      root: updateSplitSizes(state.root, splitId, sizes),
    }));
  },

  updateRequest: (tabId, patch) => {
    set((state) => ({
      root: updateTabInTree(state.root, tabId, (tab) => ({
        ...tab,
        request: { ...tab.request, ...patch },
        isDirty: true,
      })),
    }));
  },

  setResponse: (tabId, response) => {
    set((state) => ({
      root: updateTabInTree(state.root, tabId, (tab) => ({
        ...tab,
        response,
      })),
    }));
  },

  markDirty: (tabId) => {
    set((state) => ({
      root: updateTabInTree(state.root, tabId, (tab) => ({ ...tab, isDirty: true })),
    }));
  },

  markClean: (tabId) => {
    set((state) => ({
      root: updateTabInTree(state.root, tabId, (tab) => ({ ...tab, isDirty: false })),
    }));
  },

  reset: () => set(createInitialState()),
}));

// Helper: update a specific tab anywhere in the tree
function updateTabInTree(node: PaneNode, tabId: string, updater: (tab: Tab) => Tab): PaneNode {
  if (node.type === 'leaf') {
    const tabIndex = node.tabs.findIndex((t) => t.id === tabId);
    if (tabIndex === -1) return node;
    const newTabs = [...node.tabs];
    newTabs[tabIndex] = updater(newTabs[tabIndex]);
    return { ...node, tabs: newTabs };
  }
  return {
    ...node,
    children: [
      updateTabInTree(node.children[0], tabId, updater),
      updateTabInTree(node.children[1], tabId, updater),
    ],
  };
}

function updateSplitSizes(node: PaneNode, splitId: string, sizes: [number, number]): PaneNode {
  if (node.type === 'split' && node.id === splitId) {
    return { ...node, sizes };
  }
  if (node.type === 'split') {
    return {
      ...node,
      children: [
        updateSplitSizes(node.children[0], splitId, sizes),
        updateSplitSizes(node.children[1], splitId, sizes),
      ],
    };
  }
  return node;
}
```
