# Monaco → CM6 Migration — Plan 10: MultiLineEditor + Body/Response Migration

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Create the `MultiLineEditor` React component and migrate `BodyEditor` and `ResponseBodyViewer` from lazy-loaded Monaco to lazy-loaded CM6.

**Architecture:** `MultiLineEditor` is a React wrapper around CM6's `EditorView` with `basicSetup` (line numbers, folding, bracket matching, search, undo/redo), dynamic language extensions, and optional variable highlighting + hover tooltips. It uses the same controlled-value pattern as `SingleLineEditor`. Call sites lazy-load it via `React.lazy()` exactly as they currently lazy-load `MonacoWrapper`.

**Tech Stack:** CM6 (`basicSetup`, `EditorView`), React 18, extensions from Plans 01–05 + Plan 09

**Spec:** `docs/superpowers/specs/2026-04-17-monaco-to-cm6-migration-design.md` — "MultiLineEditor" and "Site 1/2"

**Depends on:** Plan 09

- [ ] **Step 1: Create the component**

Create `src/components/editor/MultiLineEditor.tsx`:

```typescript
import { EditorState } from '@codemirror/state';
import { EditorView } from '@codemirror/view';
import { basicSetup } from 'codemirror';
import { useEffect, useRef } from 'react';
import type { VariableScopeEntry } from '@/lib/url-variables';
import { cn } from '@/lib/utils';
import {
  rocketTheme,
  rocketThemeDark,
  variableContextFacet,
  setVariableContextEffect,
  variableHighlight,
} from './extensions';
import {
  detectLanguage,
  getLanguageExtension,
} from './extensions/language-detect';
import { multiLineTheme } from './extensions/multi-line-theme';
import { variableHoverTooltip } from './extensions/variable-hover';

export interface MultiLineEditorProps {
  /** Current text content (controlled). */
  value: string;
  /** Called on every content change. Omit for read-only display. */
  onChange?: (value: string) => void;
  /** CM6 language identifier: 'json' | 'xml' | 'html' | 'javascript' | 'css' | 'yaml' | 'plaintext'. */
  language?: string;
  /** Auto-detect language from body mode (json/xml/text). */
  bodyMode?: string;
  /** Auto-detect language from Content-Type header. */
  contentType?: string;
  /** Read-only mode. */
  readOnly?: boolean;
  /** CSS height. Default: '300px'. */
  height?: string;
  /** Variable context for {{var}} highlighting + hover tooltips. */
  variableContext?: Map<string, VariableScopeEntry>;
}

/**
 * Multi-line CodeMirror v6 editor with syntax highlighting, line numbers,
 * code folding, search, and optional variable highlighting.
 *
 * Drop-in replacement for MonacoWrapper. Supports the same props interface.
 */
export function MultiLineEditor({
  value,
  onChange,
  language,
  bodyMode,
  contentType,
  readOnly = false,
  height = '300px',
  variableContext,
}: MultiLineEditorProps) {
  const containerRef = useRef<HTMLDivElement>(null);
  const viewRef = useRef<EditorView | null>(null);
  const onChangeRef = useRef(onChange);
  onChangeRef.current = onChange;
  const isSyncingRef = useRef(false);

  // Resolve language: explicit prop > bodyMode > contentType > plaintext
  const resolvedLang = language ?? detectLanguage(bodyMode, contentType);

  // Build extensions. This is intentionally not memoized because the
  // editor is recreated when language/readOnly change.
  useEffect(() => {
    if (!containerRef.current) return;

    const extensions = [
      basicSetup,
      rocketTheme,
      rocketThemeDark,
      multiLineTheme,
      EditorView.lineWrapping,
    ];

    // Language extension
    const langExt = getLanguageExtension(resolvedLang);
    if (langExt) extensions.push(langExt);

    // Variable highlighting + hover (only when context provided)
    if (variableContext) {
      extensions.push(
        variableContextFacet.of(variableContext),
        variableHighlight(),
        variableHoverTooltip(),
      );
    }

    // Read-only mode
    if (readOnly) {
      extensions.push(
        EditorState.readOnly.of(true),
        EditorView.editable.of(false),
      );
    }

    // onChange listener
    extensions.push(
      EditorView.updateListener.of((update) => {
        if (update.docChanged && !isSyncingRef.current) {
          onChangeRef.current?.(update.state.doc.toString());
        }
      }),
    );

    const state = EditorState.create({
      doc: value,
      extensions,
    });

    const view = new EditorView({
      state,
      parent: containerRef.current,
    });

    viewRef.current = view;

    return () => {
      view.destroy();
      viewRef.current = null;
    };
    // Recreate editor when language or readOnly changes.
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [resolvedLang, readOnly]);

  // Sync props.value → editor when it changes externally.
  useEffect(() => {
    const view = viewRef.current;
    if (!view) return;
    const currentDoc = view.state.doc.toString();
    if (currentDoc !== value) {
      isSyncingRef.current = true;
      view.dispatch({
        changes: { from: 0, to: currentDoc.length, insert: value },
      });
      isSyncingRef.current = false;
    }
  }, [value]);

  // Update variable context facet reactively.
  useEffect(() => {
    const view = viewRef.current;
    if (!view || !variableContext) return;
    view.dispatch({
      effects: setVariableContextEffect.of(variableContext),
    });
  }, [variableContext]);

  return (
    <div
      ref={containerRef}
      className={cn('overflow-hidden border-0')}
      style={{ height }}
    />
  );
}
```
