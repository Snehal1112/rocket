# Monaco → CM6 Migration — Plan 09: Language + Theme + Hover Extensions

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Install CM6 language packages and create the three extensions needed for the multi-line editor: language detection, multi-line theme additions, and variable hover tooltips.

- [ ] **Step 1: Create the language detection module**

Create `src/components/editor/extensions/language-detect.ts`:

```typescript
import type { Extension } from '@codemirror/state';
import { json } from '@codemirror/lang-json';
import { xml } from '@codemirror/lang-xml';
import { html } from '@codemirror/lang-html';
import { javascript } from '@codemirror/lang-javascript';
import { css } from '@codemirror/lang-css';
import { yaml } from '@codemirror/lang-yaml';

/**
 * Returns a CM6 LanguageSupport extension for the given language identifier.
 * Returns null for plaintext (no language extension needed).
 */
export function getLanguageExtension(lang: string): Extension | null {
  switch (lang) {
    case 'json':
      return json();
    case 'xml':
      return xml();
    case 'html':
      return html();
    case 'javascript':
      return javascript();
    case 'css':
      return css();
    case 'yaml':
      return yaml();
    default:
      return null;
  }
}

/**
 * Detects the CM6 language identifier from a body mode or Content-Type header.
 * Replaces the Monaco `detectLanguage` function in `monaco-config.ts`.
 */
export function detectLanguage(bodyMode?: string, contentType?: string): string {
  if (bodyMode === 'json' || contentType?.includes('json')) return 'json';
  if (bodyMode === 'xml' || contentType?.includes('xml')) return 'xml';
  if (bodyMode === 'text') return 'plaintext';
  if (contentType?.includes('html')) return 'html';
  if (contentType?.includes('javascript')) return 'javascript';
  if (contentType?.includes('css')) return 'css';
  if (contentType?.includes('yaml')) return 'yaml';
  return 'plaintext';
}

/**
 * Detects the CM6 language extension from a file path (by extension).
 * Used by the DiffViewer and ConflictResolver for Git file diffs.
 */
export function getLanguageExtensionForFile(filePath: string): Extension | null {
  const ext = filePath.split('.').pop()?.toLowerCase() ?? '';
  const map: Record<string, () => Extension> = {
    json: () => json(),
    js: () => javascript(),
    ts: () => javascript({ typescript: true }),
    tsx: () => javascript({ typescript: true, jsx: true }),
    jsx: () => javascript({ jsx: true }),
    yaml: () => yaml(),
    yml: () => yaml(),
    xml: () => xml(),
    html: () => html(),
    css: () => css(),
  };
  const factory = map[ext];
  return factory ? factory() : null;
}
```

- [ ] **Step 2: Create the variable hover tooltip extension**

Create `src/components/editor/extensions/variable-hover.ts`:

```typescript
import type { Extension } from '@codemirror/state';
import { hoverTooltip, type Tooltip } from '@codemirror/view';
import { variableContextFacet } from './variable-context-facet';

const VAR_REGEX = /\{\{([\w.-]+)\}\}/g;

/**
 * Find the {{variable}} token at the given document position.
 * Returns the variable name and range, or null if the position is not
 * inside a variable token.
 */
function findVarAt(
  doc: string,
  pos: number,
): { varName: string; from: number; to: number } | null {
  VAR_REGEX.lastIndex = 0;
  let match: RegExpExecArray | null;
  while ((match = VAR_REGEX.exec(doc)) !== null) {
    const from = match.index;
    const to = from + match[0].length;
    if (pos >= from && pos <= to) {
      return { varName: match[1], from, to };
    }
  }
  return null;
}

/**
 * Hover tooltip extension for {{variable}} tokens in multi-line editors.
 *
 * Displays a read-only tooltip showing:
 * - Variable name (monospace, bold)
 * - Source scope (e.g., "Environment", "Collection", "Unresolved")
 * - Resolved value (or "●●●●" for secrets, "(not set)" for unresolved)
 *
 * This is the multi-line equivalent of Monaco's registerHoverProvider.
 * Unlike the SingleLineEditor popover, this tooltip is read-only —
 * no editing of variable values from the body editor.
 */
export function variableHoverTooltip(): Extension {
  return hoverTooltip((view, pos) => {
    const doc = view.state.doc.toString();
    const token = findVarAt(doc, pos);
    if (!token) return null;

    const context = view.state.facet(variableContextFacet);
    const entry = context.get(token.varName);

    const sourceLabel = entry
      ? entry.source.charAt(0).toUpperCase() + entry.source.slice(1)
      : 'Unresolved';
    const displayValue = entry
      ? entry.secret
        ? '●●●●'
        : entry.value || '(not set)'
      : '(not set)';

    const tooltip: Tooltip = {
      pos: token.from,
      end: token.to,
      above: true,
      create: () => {
        const dom = document.createElement('div');
        dom.className = 'cm-var-hover';

        const nameLine = document.createElement('div');
        nameLine.className = 'cm-var-hover-name';
        nameLine.textContent = `{{${token.varName}}}`;
        dom.appendChild(nameLine);

        const metaLine = document.createElement('div');
        metaLine.className = 'cm-var-hover-meta';
        metaLine.textContent = `${sourceLabel}: ${displayValue}`;
        dom.appendChild(metaLine);

        return { dom };
      },
    };

    return tooltip;
  });
}
```
