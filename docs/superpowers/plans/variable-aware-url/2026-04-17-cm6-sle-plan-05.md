# CM6 SingleLineEditor — Plan 05: URL Tokens + Secret Mask

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Create the URL-specific extensions (`:pathParam` highlighting, query param highlighting, curl paste detection) and the secret mask extension for `isSecret` fields.

**Architecture:** Two optional extension factories. `urlTokens()` returns a CM6 extension that adds `:pathParam` decoration marks, query `key=value` marks, and a paste handler for curl commands. `secretMask()` returns an extension that replaces non-`{{var}}` text with `●` via `Decoration.replace`. Both are conditionally added by `SingleLineEditor` based on props.

**Tech Stack:** `@codemirror/view` (ViewPlugin, Decoration, WidgetType), `@codemirror/state`

**Spec:** `docs/superpowers/specs/2026-04-17-codemirror-single-line-editor-design.md` — sections "URL Tokens Plugin" and "Secret Mask Plugin"

```typescript
import { type Extension, RangeSetBuilder } from '@codemirror/state';
import {
  Decoration,
  type DecorationSet,
  EditorView,
  ViewPlugin,
  type ViewUpdate,
} from '@codemirror/view';
import { parseCurl, type ParsedCurl } from '@/lib/curl-parser';

const VAR_REGEX = /\{\{[\w.-]+\}\}/g;
const PATH_PARAM_REGEX = /:(\w+)/g;

interface UrlTokensConfig {
  /** Path param values for resolution. */
  pathParams?: Record<string, string>;
  /** Query param values for resolution. */
  queryParams?: Record<string, string>;
  /** Called when a curl command is pasted. */
  onCurlImport?: (parsed: ParsedCurl) => void;
}

/**
 * ViewPlugin that highlights :pathParam and query key=value tokens
 * in the URL bar. Runs after the variable highlight plugin — it only
 * processes text segments that are NOT inside {{var}} patterns.
 */
class UrlTokensPlugin {
  decorations: DecorationSet;
  private config: UrlTokensConfig;

  constructor(view: EditorView) {
    this.config = {}; // Will be set via reconfigure
    this.decorations = this.buildDecorations(view);
  }

  update(update: ViewUpdate) {
    if (update.docChanged || update.viewportChanged) {
      this.decorations = this.buildDecorations(update.view);
    }
  }

  private buildDecorations(view: EditorView): DecorationSet {
    const builder = new RangeSetBuilder<Decoration>();
    const doc = view.state.doc.toString();

    // Find all {{var}} ranges to skip
    const varRanges: Array<[number, number]> = [];
    VAR_REGEX.lastIndex = 0;
    let vm: RegExpExecArray | null;
    while ((vm = VAR_REGEX.exec(doc)) !== null) {
      varRanges.push([vm.index, vm.index + vm[0].length]);
    }

    const isInsideVar = (pos: number) =>
      varRanges.some(([s, e]) => pos >= s && pos < e);

    // Find the ? that splits path from query
    const qIdx = doc.indexOf('?');
    const pathPart = qIdx >= 0 ? doc.slice(0, qIdx) : doc;

    // Path params: :paramName in the path portion
    PATH_PARAM_REGEX.lastIndex = 0;
    let pm: RegExpExecArray | null;
    while ((pm = PATH_PARAM_REGEX.exec(pathPart)) !== null) {
      const from = pm.index;
      const to = from + pm[0].length;
      if (isInsideVar(from)) continue;

      const paramName = pm[1];
      const resolved = this.config.pathParams?.[paramName] !== undefined;
      const cssClass = resolved ? 'cm-pathparam' : 'cm-pathparam-unresolved';
      builder.add(from, to, Decoration.mark({ class: cssClass }));
    }

    // Query params: key=value segments after ?
    if (qIdx >= 0) {
      const queryStr = doc.slice(qIdx + 1);
      const pairs = queryStr.split('&');
      let offset = qIdx + 1;

      for (const pair of pairs) {
        if (!pair) {
          offset += 1; // Skip empty segment after &
          continue;
        }
        const eqIdx = pair.indexOf('=');
        if (eqIdx > 0 && !isInsideVar(offset)) {
          const keyFrom = offset;
          const keyTo = offset + eqIdx;
          builder.add(keyFrom, keyTo, Decoration.mark({ class: 'cm-querykey' }));
        }
        offset += pair.length + 1; // +1 for the & separator
      }
    }

    return builder.finish();
  }
}

/**
 * Extension for URL bar: highlights :pathParam and query key=value tokens,
 * and detects curl paste.
 */
export function urlTokens(config: UrlTokensConfig): Extension {
  const exts: Extension[] = [
    ViewPlugin.fromClass(UrlTokensPlugin, {
      decorations: (v) => v.decorations,
    }),
  ];

  // Curl paste detection
  if (config.onCurlImport) {
    const onCurlImport = config.onCurlImport;
    exts.push(
      EditorView.domEventHandlers({
        paste: (event: ClipboardEvent, _view: EditorView) => {
          const text = event.clipboardData?.getData('text/plain')?.trim();
          if (!text || !/^curl\s/i.test(text)) return false;
          event.preventDefault();
          const parsed = parseCurl(text);
          if (parsed) {
            onCurlImport(parsed);
          }
          return true;
        },
      }),
    );
  }

  return exts;
}
```

```typescript
import { type Extension, RangeSetBuilder } from '@codemirror/state';
import {
  Decoration,
  type DecorationSet,
  EditorView,
  ViewPlugin,
  type ViewUpdate,
  WidgetType,
} from '@codemirror/view';

const VAR_REGEX = /\{\{[\w.-]+\}\}/g;

/**
 * Widget that renders ● characters as a replacement for secret text.
 * The actual document text is preserved — only the visual rendering is masked.
 */
class MaskWidget extends WidgetType {
  constructor(readonly length: number) {
    super();
  }

  eq(other: MaskWidget) {
    return this.length === other.length;
  }

  toDOM() {
    const span = document.createElement('span');
    span.textContent = '●'.repeat(this.length);
    span.className = 'cm-secret-mask';
    return span;
  }

  ignoreEvent() {
    return false;
  }
}

/**
 * ViewPlugin that replaces all non-{{variable}} text with ● characters.
 * Variable tokens remain visible and colored. The document content is
 * unchanged — only the visual presentation is masked.
 *
 * This matches Bruno's MaskedEditor behavior.
 */
class SecretMaskPlugin {
  decorations: DecorationSet;

  constructor(view: EditorView) {
    this.decorations = buildMask(view);
  }

  update(update: ViewUpdate) {
    if (update.docChanged || update.viewportChanged) {
      this.decorations = buildMask(update.view);
    }
  }
}

function buildMask(view: EditorView): DecorationSet {
  const builder = new RangeSetBuilder<Decoration>();
  const doc = view.state.doc.toString();
  if (!doc) return builder.finish();

  // Find all {{var}} ranges — these stay visible.
  const varRanges: Array<[number, number]> = [];
  VAR_REGEX.lastIndex = 0;
  let match: RegExpExecArray | null;
  while ((match = VAR_REGEX.exec(doc)) !== null) {
    varRanges.push([match.index, match.index + match[0].length]);
  }

  // Replace everything outside {{var}} ranges with ● widgets.
  let pos = 0;
  for (const [start, end] of varRanges) {
    if (pos < start) {
      const len = start - pos;
      builder.add(
        pos,
        start,
        Decoration.replace({ widget: new MaskWidget(len) }),
      );
    }
    pos = end;
  }
  // Mask trailing text after last variable
  if (pos < doc.length) {
    const len = doc.length - pos;
    builder.add(
      pos,
      doc.length,
      Decoration.replace({ widget: new MaskWidget(len) }),
    );
  }

  return builder.finish();
}

/**
 * Extension that masks non-variable text with ● characters.
 * Use for secret/password fields where the raw value should be hidden
 * but variable references remain visible.
 */
export function secretMask(): Extension {
  return ViewPlugin.fromClass(SecretMaskPlugin, {
    decorations: (v) => v.decorations,
  });
}
```
