# SP9-03 — E2E Playwright Tests (fixture-based, 4 scenarios)

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

> **⚠️ Worktree** — all commands run inside `.worktrees/contract-enhancement` on branch `feat/contract-lock-enhancement`.

**Goal:** Ship 4 self-contained E2E scenarios using Option C (YAML fixture files). `beforeAll` copies a pre-built workspace into a temp directory and points the app at it. Tests are fully isolated — no dependency on UI-based setup working.

**Architecture:**
- `e2e/fixtures/test-workspace/` — a pre-built `.rocket/` directory with TestCollection + two requests
- `beforeAll` copies it to a `$TMPDIR/rocket-e2e-workspace/` and sets a Tauri env var or workspace store to use that path
- `afterEach` re-copies only the requests directory to restore mutated request files
- `afterAll` removes the temp directory

**Depends on:** SP9-02 merged.

- [ ] **Step 3: Create `e2e/helpers/fixtures.ts`**

```typescript
import { Page } from '@playwright/test'
import { cpSync, mkdirSync, rmSync, existsSync } from 'fs'
import { join } from 'path'
import { tmpdir } from 'os'

export const FIXTURE_SOURCE = join(__dirname, '../fixtures/test-workspace')
export const TEST_WORKSPACE_DIR = join(tmpdir(), 'rocket-e2e-workspace')
export const TEST_COLLECTION_DIR = join(TEST_WORKSPACE_DIR, 'collections', 'TestCollection')

/**
 * Copies the fixture workspace to a temp directory.
 * Called in test.beforeAll.
 */
export function seedTestWorkspace(): void {
  if (existsSync(TEST_WORKSPACE_DIR)) {
    rmSync(TEST_WORKSPACE_DIR, { recursive: true, force: true })
  }
  mkdirSync(TEST_WORKSPACE_DIR, { recursive: true })
  cpSync(FIXTURE_SOURCE, TEST_WORKSPACE_DIR, { recursive: true })
}

/**
 * Resets only the requests directory (re-copies from fixture).
 * Called in test.afterEach to undo request mutations from tests.
 */
export function resetRequests(): void {
  const srcRequests = join(FIXTURE_SOURCE, 'collections', 'TestCollection', 'requests')
  const dstRequests = join(TEST_COLLECTION_DIR, 'requests')
  rmSync(dstRequests, { recursive: true, force: true })
  cpSync(srcRequests, dstRequests, { recursive: true })
}

/**
 * Removes the temp workspace directory.
 * Called in test.afterAll.
 */
export function cleanupTestWorkspace(): void {
  if (existsSync(TEST_WORKSPACE_DIR)) {
    rmSync(TEST_WORKSPACE_DIR, { recursive: true, force: true })
  }
}

/**
 * Tells the running Tauri app to switch to the test workspace.
 * Calls the switch_workspace Tauri command via page.evaluate.
 */
export async function activateTestWorkspace(page: Page): Promise<void> {
  // Check what the actual switch_workspace command accepts:
  //   grep -n "switch_workspace\|open_workspace" src-tauri/src/commands/ -r | head -5
  await page.evaluate(
    async (workspacePath: string) => {
      await (window as any).__TAURI__.core.invoke('switch_workspace', {
        path: workspacePath,
      })
    },
    TEST_WORKSPACE_DIR,
  )
  // Wait for the sidebar to reflect the new workspace
  await page.waitForTimeout(500)
}
```

- [ ] **Step 5: Create `e2e/contracts.spec.ts` with smoke test**

```typescript
import { test, expect, Page } from '@playwright/test'
import {
  seedTestWorkspace,
  resetRequests,
  cleanupTestWorkspace,
  activateTestWorkspace,
  TEST_WORKSPACE_DIR,
} from './helpers/fixtures'

// ─── Fixture lifecycle ───────────────────────────────────

test.beforeAll(() => {
  seedTestWorkspace()
})

test.afterAll(() => {
  cleanupTestWorkspace()
})

// ─── Helpers ────────────────────────────────────────────

async function setup(page: Page) {
  await activateTestWorkspace(page)
  // Wait for collection to appear in sidebar
  await expect(
    page.getByTestId('collection-item-TestCollection'),
  ).toBeVisible({ timeout: 8000 })
}

async function openContractsTab(page: Page) {
  // Prefer direct UI action over hotkey (hotkey depends on ⌘L being wired)
  await page.getByTestId('collection-item-TestCollection').click({ button: 'right' })
  await page.getByRole('menuitem', { name: 'View contracts' }).click()
  await expect(
    page.getByRole('heading', { name: 'Contracts' }),
  ).toBeVisible({ timeout: 5000 })
}

async function createPublishedContract(page: Page, name = 'Payments API v1') {
  await page.getByRole('button', { name: /New contract/i }).click()
  await expect(page.getByRole('dialog', { name: 'New contract' })).toBeVisible()
  await page.getByLabel('Title').fill(name)
  await page.getByPlaceholder('Billing Team').fill('Billing')
  await page.getByPlaceholder('Platform, Mobile').fill('Platform')
  await page.getByRole('button', { name: 'Create & Publish →' }).click()
  await expect(page.getByRole('dialog')).not.toBeVisible({ timeout: 8000 })
  await expect(
    page.locator('article', { hasText: name }),
  ).toBeVisible({ timeout: 5000 })
}

async function deleteAllContracts(page: Page) {
  // Use specific selector for contract cards (data-status attribute present)
  const cards = await page.locator('article[data-status]').all()
  for (const _ of cards) {
    const card = page.locator('article[data-status]').first()
    await card.click({ button: 'right' })
    await page.getByRole('menuitem', { name: 'Delete' }).click()
    await page.getByRole('button', { name: 'Delete contract' }).click()
    await page.waitForTimeout(200)
  }
}

// ─── Smoke test — verify testids exist ─────────────────

test('smoke: testids are present', async ({ page }) => {
  await setup(page)
  await expect(page.getByTestId('collection-item-TestCollection')).toBeVisible()
  await page.getByTestId('collection-item-TestCollection').click()
  await expect(page.getByTestId('request-item-GET-payments')).toBeVisible()
  await expect(page.getByTestId('request-item-POST-payments')).toBeVisible()
})
```

```typescript
  await setup(page)
```

```typescript
test('scenario 4: pause monitoring; drift not detected while paused; resume restarts', async ({ page }) => {
  await setup(page)
  await openContractsTab(page)
  await createPublishedContract(page, 'Payments API v1')

  // Pause via context menu
  const contractCard = page.locator('article[data-status]').first()
  await contractCard.click({ button: 'right' })
  await page.getByRole('menuitem', { name: 'Pause monitoring' }).click()
  await expect(
    page.locator('article[data-status="paused"]'),
  ).toBeVisible({ timeout: 5000 })

  // Mutate a request while paused
  await page.getByTestId('collection-item-TestCollection').click()
  await page.getByTestId('request-item-GET-payments').click()

  const methodEl = page.locator('[data-testid="method-selector"]')
    .or(page.getByRole('combobox', { name: /method/i }))
    .or(page.getByLabel('HTTP method'))
  await methodEl.first().selectOption('PUT')
  await page.keyboard.press('Meta+s')

  // Open Contracts tab and sync — should NOT show drift (paused)
  await openContractsTab(page)
  await page.getByRole('button', { name: 'Sync' }).click()
  await page.waitForTimeout(1500)  // allow recompute to settle

  await expect(page.locator('article[data-status="paused"]')).toBeVisible()
  await expect(page.locator('article[data-status="drift"]')).not.toBeVisible()
  await expect(page.locator('article[data-status="breach"]')).not.toBeVisible()

  // Resume via context menu
  // Use specific card selector (not .first() which could match wrong card after status change)
  const pausedCard = page.locator('article[data-status="paused"]', { hasText: 'Payments API v1' })
  await pausedCard.click({ button: 'right' })
  await page.getByRole('menuitem', { name: 'Resume' }).click()

  // Sync to trigger recompute after resume
  await page.getByRole('button', { name: 'Sync' }).click()

  // Card should now show drift (PUT change was made while paused)
  await expect(
    page.locator('article[data-status="drift"], article[data-status="breach"]'),
  ).toBeVisible({ timeout: 10_000 })
})
```
